# ============================================================
# Base de indicadores macro para los 27 países del estudio
# Nivel país (27 filas, ctrcode en ISO3). Criterio temporal: dato más reciente
# disponible desde 2015, registrando el año efectivamente usado por país e indicador.
# Cada indicador va en escala original y estandarizado entre los 27 países (_z).
# Todos se descargan por API; no se escriben valores a mano ni se imputa.
#
# Fuentes:
#   - Gini (ingreso disponible):  World Bank, WDI, SI.POV.GINI
#   - PIB per cápita (PPP, US$ internacionales constantes 2021):
#                                 World Bank, WDI, NY.GDP.PCAP.PP.KD
#   - Tasa de desempleo (% fuerza laboral, estimación OIT):
#                                 World Bank, WDI, SL.UEM.TOTL.ZS
#   - Empleo en la industria (% del empleo total, estimación OIT):
#                                 World Bank, WDI, SL.IND.EMPL.ZS
#   - Gasto social público total (% del PIB):
#                                 OECD SOCX (DSD_SOCX_AGG@DF_SOCX_AGG), PROGRAMME_TYPE _T
#   - Gasto público en políticas activas del mercado de trabajo (% del PIB), dividido por
#     la tasa de desempleo del mismo año (WDI):
#                                 OECD SOCX (DSD_SOCX_AGG@DF_SOCX_AGG), PROGRAMME_TYPE TP60
#   - Rigidez de la protección del empleo, contratos indefinidos (despidos individuales
#     y colectivos), versión 4, escala 0 a 6:
#                                 OECD EPL (DSD_EPL@DF_EPL), MEASURE EPL_OV, VERSION4
#   - Cobertura efectiva de las prestaciones por desempleo (ODS 1.3.1: % de personas
#     desempleadas que reciben prestaciones monetarias por desempleo), indicador de la
#     compensación. Datos hasta 2024:
#                                 OIT, ILOSTAT (ILO,DF_SDG_0131_SEX_SOC_RT), MEASURE SDG_0131_RT,

# No se incluyen, porque no cubren los 27 países:
#   - Generosidad del seguro de desempleo (SOCX TP71, Unemployment, dividido por la tasa
#     de desempleo): México no tiene dato en ningún año (estado M, "data cannot exist") y

  country   = iso_map$iso2c,
  indicator = c(gini    = "SI.POV.GINI",
                gdp_pc  = "NY.GDP.PCAP.PP.KD",
                unemp   = "SL.UEM.TOTL.ZS",
                ind_emp = "SL.IND.EMPL.ZS"),
  start = 2015, end = 2024

# Dato más reciente no perdido por país e indicador, con su año
wdi_latest <- wdi_raw %>%
  tidyr::pivot_longer(cols = c(gini, gdp_pc, unemp, ind_emp),
                      names_to = "indicator", values_to = "value") %>%
  dplyr::filter(!is.na(value)) %>%
  dplyr::group_by(iso2c, indicator) %>%
  dplyr::slice_max(year, n = 1, with_ties = FALSE) %>%
  dplyr::ungroup() %>%

  dplyr::slice_max(socx_year, n = 1, with_ties = FALSE) %>%
